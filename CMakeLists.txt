cmake_minimum_required(VERSION 3.16)
project(TreeExplorer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(bst_impl BST/BST.cpp)
add_library(rb_impl RB/RedBlackTree.cpp)
# The legacy tree implementations each declare their own Node class.
# Keep them in separate executables.
add_executable(bst BST/main.cpp)
target_link_libraries(bst PRIVATE bst_impl)
add_executable(rbtree RB/main.cpp)
target_link_libraries(rbtree PRIVATE rb_impl)
foreach(target bst_impl rb_impl bst rbtree)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_executable(bst_tests tests/trees.cpp)
    target_link_libraries(bst_tests PRIVATE bst_impl)
    add_test(NAME bst_properties COMMAND bst_tests)
    add_executable(rb_tests tests/trees.cpp)
    target_compile_definitions(rb_tests PRIVATE TEST_RB)
    target_link_libraries(rb_tests PRIVATE rb_impl)
    add_test(NAME rb_properties COMMAND rb_tests)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME console_smoke COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/console.py $<TARGET_FILE:bst> $<TARGET_FILE:rbtree>)
    endif()
endif()
